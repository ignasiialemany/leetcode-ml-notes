Temperature sampling

Temperature sampling

Insight

sample from softmax(z/T). Small T stretches gaps (near-greedy); large T flattens. Clamp tiny T to argmax.

Code

import numpy as np

def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    return e / e.sum()

def sample(z, T, rng):
    z = np.asarray(z, dtype=np.float64)
    if T <= 1e-8:
        return int(np.argmax(z))
    return int(rng.choice(len(z), p=softmax(z / T)))
