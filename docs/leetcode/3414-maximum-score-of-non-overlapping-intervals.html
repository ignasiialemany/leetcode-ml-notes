3414. Maximum Score of Non-overlapping Intervals

Weighted intervals · pick ≤k

Insight

\(k \le 4\) and \(n \le 5\cdot10^4\): sort by end, binary-search the previous interval with \(\mathrm{end} < \mathrm{start}\) (shared endpoint overlaps), then DP \(\mathrm{take}[i][t]\) for \(t=1..k\). Keep \((\mathrm{score}, \mathrm{sorted\ indices})\) and break ties by lexicographically smaller index tuples.

Visual

Example 1: indices \(2\) and \(3\) give weight \(5+3=8\). Note \([1,5]\) blocks both \([1,3]\) and \([4,5]\) because closed intervals share points.

Sort by end → DP with at most 4 picks → lex-smallest index list among max-score sets.

Code

from bisect import bisect_left
from typing import List, Optional, Tuple


class Solution:
    def maximumWeight(self, intervals: List[List[int]]) -> List[int]:
        n = len(intervals)
        arr = [
            (intervals[i][1], intervals[i][0], intervals[i][2], i) for i in range(n)
        ]
        arr.sort()
        ends = [r for r, _, _, _ in arr]
        prev = [bisect_left(ends, arr[i][1]) - 1 for i in range(n)]

        NEG = -(10**30)
        take: List[List[Tuple[int, Optional[Tuple[int, ...]]]]] = [
            [(NEG, None) for _ in range(5)] for _ in range(n)
        ]
        prefix: List[List[Tuple[int, Optional[Tuple[int, ...]]]]] = [
            [(NEG, None) for _ in range(5)] for _ in range(n)
        ]

        def better(a, b):
            sa, ia = a
            sb, ib = b
            if ia is None:
                return b
            if ib is None:
                return a
            if sa != sb:
                return a if sa > sb else b
            return a if ia < ib else b

        for i in range(n):
            _, _, w, idx = arr[i]
            take[i][1] = (w, (idx,))
            p = prev[i]
            if p >= 0:
                for k in range(2, 5):
                    sc, idxs = prefix[p][k - 1]
                    if idxs is not None:
                        take[i][k] = (sc + w, tuple(sorted(idxs + (idx,))))
            for k in range(1, 5):
                prefix[i][k] = (
                    take[i][k] if i == 0 else better(take[i][k], prefix[i - 1][k])
                )

        best: Tuple[int, Tuple[int, ...]] = (0, ())
        for k in range(1, 5):
            sc, idxs = prefix[n - 1][k]
            if idxs is not None:
                cand = (sc, idxs)
                if cand[0] > best[0] or (cand[0] == best[0] and cand[1] < best[1]):
                    best = cand
        return list(best[1])
